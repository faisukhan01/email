    50→function parseVariables(raw: string): string[] {
    51→  try {
    52→    const parsed = JSON.parse(raw);
    53→    return Array.isArray(parsed) ? parsed.map(String) : [];
    54→  } catch {
    55→    return [];
    56→  }
    57→}
    58→
    59→function deriveVariables(text: string): string[] {
    60→  const out = new Set<string>();
    61→  for (const m of text.matchAll(/\{\{\s*(\w+)\s*\}\}/g)) out.add(m[1]);
    62→  return [...out];
    63→}
    64→
    65→function renderTokens(text: string): string {
    66→  return text
    67→    .replace(/\{\{\s*first_name\s*\}\}/g, "Alex")
    68→    .replace(/\{\{\s*last_name\s*\}\}/g, "Morgan")
    69→    .replace(/\{\{\s*full_name\s*\}\}/g, "Alex Morgan")
    70→    .replace(/\{\{\s*email\s*\}\}/g, "alex@example.com")
    71→    .replace(/\{\{\s*company\s*\}\}/g, "Acme Inc")
    72→    .replace(/\{\{\s*(\w+)\s*\}\}/g, "—");
    73→}
    74→
    75→function truncateText(s: string, n: number): string {
    76→  return s.length > n ? `${s.slice(0, n)}…` : s;
    77→}

   397→// ─── Broadcast dialog ────────────────────────────────────────
   398→
   399→function BroadcastDialog({
   400→  open, lists, onOpenChange, onSaved,
   401→}: {
   402→  open: boolean; lists: { id: string; name: string }[];
   403→  onOpenChange: (v: boolean) => void; onSaved: () => void;
   404→}) {
   405→  const { toast } = useToast();
   406→  const [name, setName] = useState("");
   407→  const [bodyText, setBodyText] = useState("");
   408→  const [listId, setListId] = useState("");
   409→  const [busy, setBusy] = useState(false);
   410→
   411→  useMemo(() => {
   412→    if (open) { setName(""); setBodyText(""); setListId(""); }
   413→  }, [open]);
   414→
   415→  async function save() {
   416→    if (!name.trim()) {
   417→      toast({ title: "Name required", variant: "destructive" });
   418→      return;
   419→    }
   420→    if (!bodyText.trim()) {
   421→      toast({ title: "Message required", description: "Write the WhatsApp message to send.", variant: "destructive" });
   422→      return;
   423→    }
   424→    setBusy(true);
   425→    try {
   426→      await api.post("/api/campaigns", {
   427→        name: name.trim(),
   428→        channel: "whatsapp",
   429→        bodyText: bodyText.trim(),
   430→        audienceType: "list",
   431→        listId: listId || undefined,
   432→      });
   433→      onSaved();
   434→    } catch {
   435→      toast({ title: "Could not create broadcast", variant: "destructive" });
   436→    } finally {
   437→      setBusy(false);
   438→    }
   439→  }
   440→
   441→  return (
   442→    <Dialog open={open} onOpenChange={onOpenChange}>
   443→      <DialogContent className="sm:max-w-lg">
   444→        <DialogHeader>
   445→          <DialogTitle>New WhatsApp broadcast</DialogTitle>
   446→          <DialogDescription>Quick one-off message to a list. You can review before sending.</DialogDescription>
   447→        </DialogHeader>
   448→
   449→        <div className="space-y-3">
   450→          <div className="space-y-1.5">
   451→            <Label htmlFor="bc-name">Broadcast name</Label>
   452→            <Input id="bc-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Flash sale announcement" maxLength={140} />
   453→          </div>
   454→          <div className="space-y-1.5">
   455→            <Label htmlFor="bc-list">Audience list</Label>
   456→            <Select value={listId} onValueChange={setListId}>
   457→              <SelectTrigger aria-label="Audience list"><SelectValue placeholder="Choose a list" /></SelectTrigger>
   458→              <SelectContent>
   459→                {lists.map((l) => <SelectItem key={l.id} value={l.id}>{l.name}</SelectItem>)}
   460→              </SelectContent>
   461→            </Select>
   462→          </div>
   463→          <div className="space-y-1.5">
   464→            <Label htmlFor="bc-body">Message</Label>
   465→            <Textarea
   466→              id="bc-body" value={bodyText} rows={5} maxLength={1024}
   467→              onChange={(e) => setBodyText(e.target.value)}
   468→              placeholder="Hi {{first_name}}! ⚡ Flash sale starts today — 24 hours only."
   469→            />
   470→            <div className="flex items-center justify-between text-xs text-muted-foreground">
   471→              <span>Tokens like <span className="font-mono">{"{{first_name}}"}</span> are personalized per contact.</span>
   472→              <span className={bodyText.length > 950 ? "font-medium text-amber-600 dark:text-amber-400" : ""}>
   473→                {bodyText.length}/1024

   499→function WaTemplateDialog({
   500→  open, template, onOpenChange, onSaved,
   501→}: {
   502→  open: boolean; template: WaTemplate | null;
   503→  onOpenChange: (v: boolean) => void; onSaved: () => void;
   504→}) {
   505→  const { toast } = useToast();
   506→  const [name, setName] = useState("");
   507→  const [bodyText, setBodyText] = useState("");
   508→  const [busy, setBusy] = useState(false);
   509→
   510→  useMemo(() => {
   511→    if (open) {
   512→      setName(template?.name ?? "");
   513→      setBodyText(template?.bodyText ?? "");
   514→    }
   515→  }, [template, open]);
   516→
   517→  const variables = useMemo(() => deriveVariables(bodyText), [bodyText]);
   518→
   519→  function insertToken(token: string) {
   520→    setBodyText((prev) => `${prev}{{${token}}}`);
   521→  }
   522→
   523→  async function save() {
   524→    if (!name.trim()) {
   525→      toast({ title: "Name required", variant: "destructive" });
   526→      return;
   527→    }
   528→    if (!bodyText.trim()) {
   529→      toast({ title: "Message required", variant: "destructive" });
   530→      return;
   531→    }
   532→    setBusy(true);
   533→    try {
   534→      const payload = { name: name.trim(), channel: "whatsapp", bodyText: bodyText.trim(), variables };
   535→      if (template) await api.patch(`/api/templates/${template.id}`, payload);
   536→      else await api.post("/api/templates", payload);
   537→      onSaved();
   538→    } catch {
   539→      toast({ title: "Could not save template", variant: "destructive" });
   540→    } finally {
   541→      setBusy(false);
   542→    }
   543→  }
   544→
   545→  return (
   546→    <Dialog open={open} onOpenChange={onOpenChange}>
   547→      <DialogContent className="sm:max-w-lg">
   548→        <DialogHeader>
   549→          <DialogTitle>{template ? "Edit template" : "New WhatsApp template"}</DialogTitle>
   550→          <DialogDescription>Variables are detected automatically from your tokens.</DialogDescription>
   551→        </DialogHeader>
   552→
   553→        <div className="space-y-3">
   554→          <div className="space-y-1.5">
   555→            <Label htmlFor="watpl-name">Template name</Label>
   556→            <Input id="watpl-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Welcome message" maxLength={120} />
   557→          </div>
   558→          <div className="space-y-1.5">
   559→            <Label htmlFor="watpl-body">Message body</Label>
   560→            <div className="flex flex-wrap gap-1" aria-label="Insert variable token">
   561→              {TOKEN_BUTTONS.map((t) => (
   562→                <button
   563→                  key={t} type="button"
   564→                  className="rounded border px-1.5 py-1 font-mono text-[10px] text-muted-foreground hover:bg-muted"
   565→                  onClick={() => insertToken(t)}
   566→                >
   567→                  +{t}
   568→                </button>
   569→              ))}
   570→            </div>
   571→            <Textarea
   572→              id="watpl-body" value={bodyText} rows={6} maxLength={1024}
   573→              onChange={(e) => setBodyText(e.target.value)}
   574→              placeholder="Hi {{first_name}}, thanks for reaching out! How can we help?"
   575→            />
   576→            <p className="text-xs text-muted-foreground">{bodyText.length}/1024 characters</p>
   577→          </div>
   578→          {variables.length > 0 && (
   579→            <div className="flex flex-wrap items-center gap-1.5">
   580→              <span className="text-xs text-muted-foreground">Detected variables:</span>
   581→              {variables.map((v) => (
   582→                <Badge key={v} variant="secondary" className="font-mono text-[10px]">{`{{${v}}}`}</Badge>
   583→              ))}
   584→            </div>
   585→          )}
   586→          <div className="flex justify-center rounded-lg bg-stone-50 p-3 dark:bg-stone-900">
   587→            <div className="max-w-[260px] rounded-2xl rounded-br-sm bg-lime-100 p-3 text-sm shadow dark:bg-lime-900">
   588→              {renderTokens(bodyText) || <span className="text-stone-400">Message preview…</span>}
   589→              <p className="mt-1 text-right text-[10px] text-stone-500">delivered ✓✓</p>
   590→            </div>
   591→          </div>
   592→        </div>
   593→
   594→        <DialogFooter>
   595→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   596→          <Button onClick={save} disabled={busy} className="bg-teal-600 hover:bg-teal-700">
   597→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   598→            {template ? "Save changes" : "Create template"}
   599→          </Button>
   600→        </DialogFooter>
   601→      </DialogContent>
   602→    </Dialog>
   603→  );
   604→}
