    74→// ─── CSV parsing (minimal quoted-field support) ──────────────
    75→
    76→const HEADER_ALIASES: Record<string, string> = {
    77→  email: "email", emailaddress: "email", mail: "email",
    78→  firstname: "firstName", fname: "firstName", first: "firstName",
    79→  lastname: "lastName", lname: "lastName", last: "lastName",
    80→  phone: "phone", mobile: "phone", phonenumber: "phone",
    81→  company: "company", organization: "company", org: "company",
    82→  country: "country", city: "city",
    83→  jobtitle: "jobTitle", title: "jobTitle", role: "jobTitle",
    84→};

    86→function splitCsvLine(line: string): string[] {
    87→  const cells: string[] = [];
    88→  let cur = "";
    89→  let inQuotes = false;
    90→  for (let i = 0; i < line.length; i++) {
    91→    const ch = line[i];
    92→    if (inQuotes) {
    93→      if (ch === '"') {
    94→        if (line[i + 1] === '"') { cur += '"'; i++; } else { inQuotes = false; }
    95→      } else {
    96→        cur += ch;
    97→      }
    98→    } else if (ch === '"') {
    99→      inQuotes = true;
   100→    } else if (ch === ",") {
   101→      cells.push(cur); cur = "";
   102→    } else {
   103→      cur += ch;
   104→    }
   105→  }
   106→  cells.push(cur);
   107→  return cells.map((c) => c.trim());
   108→}
   109→
   110→function parseCsv(text: string): { rows: Record<string, string>[]; error: string | null } {
   111→  const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean);
   112→  if (lines.length < 2) return { rows: [], error: "Paste at least a header row and one data row." };
   113→  const headers = splitCsvLine(lines[0]).map((h) => HEADER_ALIASES[h.toLowerCase().replace(/[^a-z0-9]/g, "")] ?? "");
   114→  if (!headers.includes("email")) {
   115→    return { rows: [], error: "No email column found. Include a header named 'email'." };
   116→  }
   117→  const rows: Record<string, string>[] = [];
   118→  for (const line of lines.slice(1)) {
   119→    const cells = splitCsvLine(line);
   120→    const row: Record<string, string> = {};
   121→    headers.forEach((h, i) => {
   122→      if (h && FORM_FIELDS.includes(h as (typeof FORM_FIELDS)[number])) row[h] = cells[i] ?? "";
   123→    });
   124→    if (row.email) rows.push(row);
   125→  }
   126→  return { rows, error: rows.length === 0 ? "No data rows with an email found." : null };
   127→}

   722→function TagDialog({
   723→  open, mode, contact, onOpenChange,
   724→}: {
   725→  open: boolean; mode: "add" | "remove"; contact: Contact | null;
   726→  onOpenChange: (o: boolean) => void;
   727→}) {
   728→  const qc = useQueryClient();
   729→  const { toast } = useToast();
   730→  const [tag, setTag] = useState("");
   731→
   732→  const patchTags = useMutation({
   733→    mutationFn: (vars: { id: string; tags: string[]; verb: string }) =>
   734→      api.patch(`/api/contacts/${vars.id}`, { tags: vars.tags }).then(() => vars.verb),
   735→    onSuccess: (verb) => {
   736→      qc.invalidateQueries({ queryKey: ["contacts"] });
   737→      qc.invalidateQueries({ queryKey: ["contact"] });
   738→      toast({ title: verb === "added" ? "Tag added" : "Tag removed" });
   739→    },
   740→    onError: (err) => toast({ title: "Could not update tags", description: describeError(err), variant: "destructive" }),
   741→  });
   742→
   743→  function addTag() {
   744→    const t = tag.trim().slice(0, 40);
   745→    if (!t || !contact) return;
   746→    if (contact.tags.includes(t)) {
   747→      toast({ title: "Tag already exists on this contact", variant: "destructive" });
   748→      return;
   749→    }
   750→    patchTags.mutate({ id: contact.id, tags: [...contact.tags, t].slice(0, 30), verb: "added" });
   751→    setTag("");
   752→  }
   753→
   754→  function removeTag(t: string) {
   755→    if (!contact) return;
   756→    patchTags.mutate({ id: contact.id, tags: contact.tags.filter((x) => x !== t), verb: "removed" });
   757→  }
   758→
   759→  return (
   760→    <Dialog open={open} onOpenChange={(o) => { if (!o) setTag(""); onOpenChange(o); }}>
   761→      <DialogContent className="sm:max-w-sm">
   762→        <DialogHeader>
   763→          <DialogTitle>{mode === "add" ? "Add tag" : "Remove tag"}</DialogTitle>
   764→          <DialogDescription className="truncate">
   765→            {contact ? contactName(contact) : ""}
   766→          </DialogDescription>
   767→        </DialogHeader>
   768→
   769→        {mode === "add" ? (
   770→          <div className="space-y-3">
   771→            <div className="space-y-1.5">
   772→              <Label htmlFor="row-tag">Tag name</Label>
   773→              <Input
   774→                id="row-tag"
   775→                value={tag}
   776→                onChange={(e) => setTag(e.target.value)}
   777→                onKeyDown={(e) => { if (e.key === "Enter") { e.preventDefault(); addTag(); } }}
   778→                placeholder="vip"
   779→                maxLength={40}
   780→                autoFocus
   781→              />
   782→            </div>
   783→            {contact && contact.tags.length > 0 && (
   784→              <div className="flex flex-wrap gap-1">
   785→                {contact.tags.map((t) => (
   786→                  <Badge key={t} variant="secondary">{t}</Badge>
   787→                ))}
   788→              </div>
   789→            )}
   790→          </div>
   791→        ) : (
   792→          <div className="flex flex-wrap gap-1.5">
   793→            {contact?.tags.map((t) => (
   794→              <Badge key={t} variant="secondary" className="gap-1 py-1 pl-2.5">
   795→                {t}
   796→                <button
   797→                  onClick={() => removeTag(t)}
   798→                  className="rounded-full p-0.5 hover:bg-stone-200 dark:hover:bg-stone-700"
   799→                  aria-label={`Remove tag ${t}`}
   800→                  disabled={patchTags.isPending}
   801→                >
   802→                  <X className="h-3 w-3" aria-hidden />
   803→                </button>
   804→              </Badge>
   805→            ))}
   806→          </div>
   807→        )}
   808→
   809→        <DialogFooter>
   810→          {mode === "add" ? (
   811→            <Button onClick={addTag} disabled={patchTags.isPending || !tag.trim()} className="bg-teal-600 hover:bg-teal-700">
   812→              {patchTags.isPending && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   813→              Add tag
   814→            </Button>
   815→          ) : (
   816→            <Button variant="outline" onClick={() => onOpenChange(false)}>Done</Button>
   817→          )}
   818→        </DialogFooter>
   819→      </DialogContent>
   820→    </Dialog>
   821→  );
   822→}

   874→function ImportDialog({

   885→  const parsed = useMemo(() => parseCsv(csvText), [csvText]);
   886→
   887→  const importMutation = useMutation({
   888→    mutationFn: (vars: { contacts: Record<string, string>[]; listId?: string }) =>
   889→      api.post<ImportResult>("/api/contacts/import", {
   890→        contacts: vars.contacts.slice(0, 2000),
   891→        ...(vars.listId ? { listId: vars.listId } : {}),

   896→      qc.invalidateQueries({ queryKey: ["lists"] });

   898→        title: "Import finished",
   899→        description: `${data.created} created · ${data.updated} updated · ${data.skipped} skipped.`,
