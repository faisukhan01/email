    95→let uidCounter = 0;
    96→function uid(): string {
    97→  uidCounter += 1;
    98→  return `s${Date.now().toString(36)}${uidCounter.toString(36)}`;
    99→}
   100→
   101→function parseObj(v: unknown): Record<string, unknown> {
   102→  if (typeof v === "string") {
   103→    try { return JSON.parse(v) as Record<string, unknown>; } catch { return {}; }
   104→  }
   105→  return (v ?? {}) as Record<string, unknown>;
   106→}
   107→
   108→function truncateText(s: string, n: number): string {
   109→  return s.length > n ? `${s.slice(0, n)}…` : s;
   110→}
   111→
   112→function renderTokens(text: string): string {
   113→  return text
   114→    .replace(/\{\{\s*first_name\s*\}\}/g, "Alex")
   115→    .replace(/\{\{\s*last_name\s*\}\}/g, "Morgan")
   116→    .replace(/\{\{\s*full_name\s*\}\}/g, "Alex Morgan")
   117→    .replace(/\{\{\s*email\s*\}\}/g, "alex@example.com")
   118→    .replace(/\{\{\s*company\s*\}\}/g, "Acme Inc")
   119→    .replace(/\{\{\s*job_title\s*\}\}/g, "Product Manager")
   120→    .replace(/\{\{\s*city\s*\}\}/g, "Dubai")
   121→    .replace(/\{\{\s*country\s*\}\}/g, "UAE")
   122→    .replace(/\{\{\s*(\w+)\s*\}\}/g, "—");
   123→}
   124→
   125→function fmtDelay(n: number, unit: DelayUnit): string {
   126→  if (!n) return "no delay";
   127→  return `${n} ${n === 1 ? unit.slice(0, -1) : unit}`;
   128→}
   129→
   130→function newStep(type: StepType): BuilderStep {
   131→  return {
   132→    key: uid(), type,
   133→    delayAmount: type === "wait" ? 1 : 0,
   134→    delayUnit: "days",
   135→    subject: type === "send_email" ? "Welcome, {{first_name}}!" : "",
   136→    bodyHtml: type === "send_email" ? DEFAULT_EMAIL_HTML : "",
   137→    bodyText: type === "send_whatsapp" ? "Hi {{first_name}}! 👋 Thanks for joining us." : "",
   138→    tag: "", field: "", value: "",
   139→  };
   140→}
   141→
   142→function loadSteps(sequence: Sequence): BuilderStep[] {
   143→  return sequence.steps.map((st) => {
   144→    const cfg = parseObj(st.config);
   145→    const amount = Number(cfg.delayAmount ?? 0);
   146→    const unit = String(cfg.delayUnit ?? "days");
   147→    return {
   148→      key: st.id || uid(),
   149→      type: (STEP_META[st.type as StepType] ? st.type : "wait") as StepType,
   150→      delayAmount: Number.isFinite(amount) ? amount : 0,
   151→      delayUnit: (["minutes", "hours", "days"].includes(unit) ? unit : "days") as DelayUnit,
   152→      subject: String(cfg.subject ?? ""),
   153→      bodyHtml: String(cfg.bodyHtml ?? ""),
   154→      bodyText: String(cfg.bodyText ?? ""),
   155→      tag: String(cfg.tag ?? ""),
   156→      field: String(cfg.field ?? ""),
   157→      value: String(cfg.value ?? ""),
   158→    };
   159→  });
   160→}
   161→
   162→function stepConfig(s: BuilderStep): Record<string, unknown> {
   163→  switch (s.type) {
   164→    case "send_email": return { subject: s.subject, bodyHtml: s.bodyHtml };
   165→    case "send_whatsapp": return { bodyText: s.bodyText };
   166→    case "add_tag":
   167→    case "remove_tag": return { tag: s.tag };
   168→    case "update_field": return { field: s.field, value: s.value };
   169→    default: return {};
   170→  }
   171→}
   172→
   173→function stepSummary(s: BuilderStep): string {
   174→  switch (s.type) {
   175→    case "send_email": return s.subject.trim() || "(no subject)";
   176→    case "send_whatsapp": return s.bodyText.trim() ? truncateText(s.bodyText.trim(), 64) : "(empty message)";
   177→    case "wait": return `Waits ${fmtDelay(s.delayAmount, s.delayUnit)}`;
   178→    case "add_tag": return s.tag.trim() ? `#${s.tag.trim()}` : "(no tag set)";
   179→    case "remove_tag": return s.tag.trim() ? `Remove #${s.tag.trim()}` : "(no tag set)";
   180→    case "update_field": {
   181→      const label = FIELD_OPTIONS.find((f) => f.value === s.field)?.label ?? "";
   182→      return s.field ? `${label} → ${s.value.trim() || "—"}` : "(no field chosen)";
   183→    }
   184→    default: return "";
   185→  }
   186→}

   378→function SequenceBuilder({
   379→  open, sequence, lists, onOpenChange, onSaved,
   380→}: {
   381→  open: boolean; sequence: Sequence | null;
   382→  lists: { id: string; name: string }[];
   383→  onOpenChange: (v: boolean) => void; onSaved: () => void;
   384→}) {
   385→  const { toast } = useToast();
   386→  const [name, setName] = useState("");
   387→  const [description, setDescription] = useState("");
   388→  const [triggerType, setTriggerType] = useState<string>("contact_created");
   389→  const [triggerListId, setTriggerListId] = useState("");
   390→  const [triggerTag, setTriggerTag] = useState("");
   391→  const [steps, setSteps] = useState<BuilderStep[]>([]);
   392→  const [selectedKey, setSelectedKey] = useState<string | null>(null);
   393→  const [addType, setAddType] = useState("");
   394→  const [activateAfter, setActivateAfter] = useState(false);
   395→  const [previewOpen, setPreviewOpen] = useState(false);
   396→  const [busy, setBusy] = useState(false);
   397→
   398→  // Sync form when the dialog opens (render-phase sync — never inside useEffect).
   399→  useMemo(() => {
   400→    if (open && sequence) {
   401→      const cfg = parseObj(sequence.triggerConfig);
   402→      setName(sequence.name);
   403→      setDescription(sequence.description);
   404→      setTriggerType(sequence.triggerType);
   405→      setTriggerListId(typeof cfg.listId === "string" ? cfg.listId : "");
   406→      setTriggerTag(typeof cfg.tag === "string" ? cfg.tag : "");
   407→      setSteps(loadSteps(sequence));
   408→      setSelectedKey(sequence.steps[0]?.id ?? null);
   409→      setActivateAfter(false);
   410→    } else if (open) {
   411→      setName(""); setDescription("");
   412→      setTriggerType("contact_created"); setTriggerListId(""); setTriggerTag("");
   413→      const first = newStep("send_email");
   414→      setSteps([first]);
   415→      setSelectedKey(first.key);
   416→      setActivateAfter(false);
   417→    }
   418→  }, [sequence, open]);
   419→
   420→  const selected = steps.find((s) => s.key === selectedKey) ?? null;
   421→
   422→  function patchStep(key: string, patch: Partial<BuilderStep>) {
   423→    setSteps((prev) => prev.map((s) => (s.key === key ? { ...s, ...patch } : s)));
   424→  }
   425→
   426→  function addStep(type: StepType) {
   427→    if (steps.length >= 30) return;
   428→    const step = newStep(type);
   429→    setSteps((prev) => [...prev, step]);
   430→    setSelectedKey(step.key);
   431→  }
   432→
   433→  function deleteStep(key: string) {
   434→    setSteps((prev) => prev.filter((s) => s.key !== key));
   435→    if (selectedKey === key) setSelectedKey(null);
   436→  }
   437→
   438→  function moveStep(key: string, dir: -1 | 1) {
   439→    setSteps((prev) => {
   440→      const idx = prev.findIndex((s) => s.key === key);
   441→      const target = idx + dir;
   442→      if (idx < 0 || target < 0 || target >= prev.length) return prev;
   443→      const next = [...prev];
   444→      [next[idx], next[target]] = [next[target], next[idx]];
   445→      return next;
   446→    });
   447→  }
   448→
   449→  function validate(): string | null {
   450→    for (let i = 0; i < steps.length; i++) {
   451→      const s = steps[i];
   452→      if (s.type === "send_email" && !s.bodyHtml.trim()) return `Step ${i + 1}: email body is empty`;
   453→      if (s.type === "send_whatsapp" && !s.bodyText.trim()) return `Step ${i + 1}: WhatsApp message is empty`;
   454→      if ((s.type === "add_tag" || s.type === "remove_tag") && !s.tag.trim()) return `Step ${i + 1}: tag name is empty`;
   455→      if (s.type === "update_field" && !s.field) return `Step ${i + 1}: choose a field to update`;
   456→    }
   457→    return null;
   458→  }
   459→
   460→  async function save() {
   461→    if (!name.trim()) {
   462→      toast({ title: "Name required", description: "Give your sequence a name first.", variant: "destructive" });
   463→      return;
   464→    }
   465→    const problem = validate();
   466→    if (problem) {
   467→      toast({ title: "Incomplete step", description: problem, variant: "destructive" });
   468→      return;
   469→    }
   470→    setBusy(true);
   471→    try {
   472→      const payload = {
   473→        name: name.trim(),
   474→        description: description.trim(),
   475→        triggerType,
   476→        triggerConfig:
   477→          triggerType === "added_to_list"
   478→            ? { listId: triggerListId || undefined }
   479→            : triggerType === "tagged"
   480→              ? { tag: triggerTag.trim() }
   481→              : {},
   482→      };
   483→      let id = sequence?.id;
   484→      if (id) {
   485→        await api.patch(`/api/sequences/${id}`, payload);
   486→      } else {
   487→        const created = await api.post<{ sequence: { id: string } }>("/api/sequences", payload);
   488→        id = created.sequence.id;
   489→      }
   490→      await api.put(`/api/sequences/${id}/steps`, {
   491→        steps: steps.map((s) => ({
   492→          type: s.type,
   493→          name: STEP_META[s.type].label,
   494→          delayAmount: Math.max(0, Math.min(365, Math.round(s.delayAmount || 0))),
   495→          delayUnit: s.delayUnit,
   496→          config: stepConfig(s),
   497→        })),
   498→      });
   499→      if (activateAfter && id) {

   898→
   899→  const results = useQuery({
   900→    queryKey: ["contacts-search", search],

   903→  });

   905→  function onSearchChange(v: string) {
   906→    setSearchInput(v);

   913→      prev.some((x) => x.id === c.id) ? prev.filter((x) => x.id !== c.id) : [...prev, c]
   914→    );
   915→  }
   916→

   918→    mutationFn: () =>
   919→      api.post<{ enrolled: number; skipped: number }>(`/api/sequences/${sequence.id}/enroll`, {
   920→        contactIds: selected.map((c) => c.id),
   921→      }),
   922→    onSuccess: (d) => {
   923→      qc.invalidateQueries({ queryKey: ["sequences"] });
   924→      toast({
   925→        title: `Enrolled ${d.enrolled} contact${d.enrolled === 1 ? "" : "s"}`,
   926→        description: d.skipped > 0 ? `${d.skipped} were already active in this sequence.` : "Step 1 is scheduled for each of them.",
   927→      });

   929→    },

   931→  });

   934→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   935→      <DialogContent className="sm:max-w-lg">
   936→        <DialogHeader>
   937→          <DialogTitle>Enroll contacts into &quot;{sequence.name}&quot;</DialogTitle>

   939→        </DialogHeader>
   940→
   941→        <div className="space-y-3">
   942→          <Input
   943→            value={searchInput}
   944→            onChange={(e) => onSearchChange(e.target.value)}
   945→            placeholder="Search contacts by name, email or company…"
   946→            aria-label="Search contacts"
   947→          />
   948→
   949→          {selected.length > 0 && (
   950→            <div className="flex flex-wrap gap-1.5">
   951→              {selected.map((c) => (
   952→                <Badge key={c.id} variant="secondary" className="gap-1 bg-teal-100 text-teal-800 dark:bg-teal-950 dark:text-teal-300">
   953→                  {c.firstName || c.email}
   954→                  <button
   955→                    type="button"
   956→                    onClick={() => toggle(c)}
   957→                    aria-label={`Remove ${c.firstName || c.email} from selection`}
   958→                    className="rounded-full hover:bg-teal-200 dark:hover:bg-teal-900"
   959→                  >
   960→                    <X className="h-3 w-3" aria-hidden />
   961→                  </button>
   962→                </Badge>
   963→              ))}
   964→            </div>
   965→          )}
   966→
   967→          <div
   968→            className={`max-h-64 divide-y overflow-y-auto rounded-lg border ${SCROLLBAR}`}
   969→            aria-label="Contact results"
   970→          >
   971→            {results.isLoading ? (
   972→              <div className="space-y-2 p-3">
   973→                {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   974→              </div>
   975→            ) : (results.data?.contacts ?? []).length === 0 ? (
   976→              <p className="p-6 text-center text-sm text-muted-foreground">
   977→                {search ? `No contacts match "${search}".` : "No contacts found."}
   978→              </p>
   979→            ) : (
   980→              (results.data?.contacts ?? []).map((c) => {
   981→                const checked = selected.some((x) => x.id === c.id);
   982→                return (
   983→                  <label
