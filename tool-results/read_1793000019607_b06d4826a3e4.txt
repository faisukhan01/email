    57→    57→function StatBar({ stats }: { stats: Record<string, number> }) {
    58→    58→  const total = Object.values(stats).reduce((a, b) => a + b, 0);
    59→    59→  if (total === 0) return <span className="text-xs text-muted-foreground">Not sent yet</span>;
    60→    60→  const clicked = stats.clicked ?? 0;
    61→    61→  const opened = (stats.opened ?? 0) + clicked;
    62→    62→  const sent = total - (stats.bounced ?? 0) - (stats.failed ?? 0) - (stats.queued ?? 0);
    63→    63→  const pct = (n: number) => (sent > 0 ? Math.round((n / sent) * 100) : 0);
    64→    64→  return (
    65→    65→    <div className="space-y-1">
    66→    66→      <div className="flex gap-0.5">
    67→    67→        <div className="h-1.5 rounded-l bg-teal-500" style={{ width: `${Math.max(pct(sent - opened), opened < sent ? 8 : 4)}%` }} />
    68→    68→        <div className="h-1.5 bg-amber-400" style={{ width: `${Math.max(pct(opened - clicked), opened > 0 ? 6 : 0)}%` }} />
    69→    69→        <div className="h-1.5 rounded-r bg-lime-500" style={{ width: `${Math.max(pct(clicked), clicked > 0 ? 6 : 0)}%` }} />
    70→    70→      </div>
    71→    71→      <p className="text-[11px] text-muted-foreground">
    72→    72→        {formatNumber(total)} recipients · {pct(opened)}% opens · {pct(clicked)}% clicks
    73→    73→      </p>
    74→    74→    </div>
    75→    75→  );
    76→    76→}

   277→   277→function CampaignEditor({
   278→   278→  open, campaign, lists, onOpenChange, onSaved, previewMode, setPreviewMode,
   279→   279→}: {
   280→   280→  open: boolean; campaign: Campaign | null; lists: { id: string; name: string }[];
   281→   281→  onOpenChange: (v: boolean) => void; onSaved: () => void;
   282→   282→  previewMode: boolean; setPreviewMode: (v: boolean) => void;
   283→   283→}) {
   284→   284→  const { toast } = useToast();
   285→   285→  const [name, setName] = useState("");
   286→   286→  const [channel, setChannel] = useState("email");
   287→   287→  const [subject, setSubject] = useState("");
   288→   288→  const [preheader, setPreheader] = useState("");
   289→   289→  const [fromName, setFromName] = useState("");
   290→   290→  const [fromEmail, setFromEmail] = useState("");
   291→   291→  const [bodyHtml, setBodyHtml] = useState(DEFAULT_HTML);
   292→   292→  const [bodyText, setBodyText] = useState("");
   293→   293→  const [audienceType, setAudienceType] = useState("list");
   294→   294→  const [listId, setListId] = useState("");
   295→   295→  const [busy, setBusy] = useState(false);
   296→   296→
   297→   297→  useMemo(() => {
   298→   298→    if (campaign) {
   299→   299→      setName(campaign.name); setChannel(campaign.channel); setSubject(campaign.subject);
   300→   300→      setPreheader(campaign.preheader); setFromName(campaign.fromName); setFromEmail(campaign.fromEmail);
   301→   301→      setBodyHtml(campaign.bodyHtml || DEFAULT_HTML); setBodyText(campaign.bodyText);
   302→   302→      setAudienceType(campaign.audienceType); setListId(campaign.listId ?? "");
   303→   303→    } else {
   304→   304→      setName(""); setChannel("email"); setSubject(""); setPreheader("");
   305→   305→      setFromName(""); setFromEmail(""); setBodyHtml(DEFAULT_HTML); setBodyText("");
   306→   306→      setAudienceType("list"); setListId("");
   307→   307→    }
   308→   308→  }, [campaign, open]);
   309→   309→
   310→   310→  const previewHtml = useMemo(
   311→   311→    () =>
   312→   312→      (bodyHtml || "<p></p>")
   313→   313→        .replace(/\{\{\s*first_name\s*\}\}/g, "Alex")
   314→   314→        .replace(/\{\{\s*last_name\s*\}\}/g, "Morgan")
   315→   315→        .replace(/\{\{\s*full_name\s*\}\}/g, "Alex Morgan")
   316→   316→        .replace(/\{\{\s*company\s*\}\}/g, "Acme Inc")
   317→   317→        .replace(/\{\{\s*(\w+)\s*\}\}/g, "—"),
   318→   318→    [bodyHtml]
   319→   319→  );
   320→   320→
   321→   321→  async function save() {
   322→   322→    if (!name.trim()) {
   323→   323→      toast({ title: "Name required", variant: "destructive" });
   324→   324→      return;
   325→   325→    }
   326→   326→    setBusy(true);
   327→   327→    try {
   328→   328→      const payload = {
   329→   329→        name, channel, subject, preheader, fromName, fromEmail,
   330→   330→        bodyHtml: channel === "email" ? bodyHtml : "",
   331→   331→        bodyText: channel === "whatsapp" ? bodyText : "",
   332→   332→        audienceType,
   333→   333→        listId: audienceType === "list" && listId ? listId : undefined,
   334→   334→      };
   335→   335→      if (campaign) {
   336→   336→        await api.patch(`/api/campaigns/${campaign.id}`, payload);
   337→   337→      } else {
   338→   338→        await api.post("/api/campaigns", payload);
   339→   339→      }
   340→   340→      onSaved();
   341→   341→    } catch {
   342→   342→      toast({ title: "Could not save campaign", variant: "destructive" });
   343→   343→    } finally {
   344→   344→      setBusy(false);
   345→   345→    }
   346→   346→  }
   347→   347→
   348→   348→  return (
   349→   349→    <Dialog open={open} onOpenChange={onOpenChange}>
   350→   350→      <DialogContent className="max-w-3xl">
   351→   351→        <DialogHeader>
   352→   352→          <DialogTitle>{campaign ? "Edit campaign" : "New campaign"}</DialogTitle>
   353→   353→          <DialogDescription>
   354→   354→            Personalize with tokens — the engine renders them per contact at send time.
   355→   355→          </DialogDescription>
   356→   356→        </DialogHeader>
   357→   357→
   358→   358→        <div className="grid gap-4 sm:grid-cols-2">
   359→   359→          <div className="space-y-3">
   360→   360→            <div className="space-y-1.5">
   361→   361→              <Label htmlFor="camp-name">Campaign name</Label>
   362→   362→              <Input id="camp-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="October Newsletter" maxLength={140} />
   363→   363→            </div>
   364→   364→
   365→   365→            <div className="grid grid-cols-2 gap-3">
   366→   366→              <div className="space-y-1.5">
   367→   367→                <Label>Channel</Label>
   368→   368→                <Select value={channel} onValueChange={setChannel}>
   369→   369→                  <SelectTrigger aria-label="Channel"><SelectValue /></SelectTrigger>
   370→   370→                  <SelectContent>
   371→   371→                    <SelectItem value="email">Email</SelectItem>
   372→   372→                    <SelectItem value="whatsapp">WhatsApp</SelectItem>
   373→   373→                  </SelectContent>
   374→   374→                </Select>
   375→   375→              </div>
   376→   376→              <div className="space-y-1.5">
   377→   377→                <Label>Audience</Label>
   378→   378→                <Select value={audienceType} onValueChange={setAudienceType}>
   379→   379→                  <SelectTrigger aria-label="Audience"><SelectValue /></SelectTrigger>
   380→   380→                  <SelectContent>
   381→   381→                    <SelectItem value="list">A list</SelectItem>
   382→   382→                    <SelectItem value="all">All contacts</SelectItem>
   383→   383→                    <SelectItem value="segment">Segment rules</SelectItem>
   384→   384→                  </SelectContent>
   385→   385→                </Select>
   386→   386→              </div>
   387→   387→            </div>
   388→   388→
   389→   389→            {audienceType === "list" && (
   390→   390→              <div className="space-y-1.5">
   391→   391→                <Label>Send to list</Label>
   392→   392→                <Select value={listId} onValueChange={setListId}>
   393→   393→                  <SelectTrigger aria-label="List"><SelectValue placeholder="Choose a list" /></SelectTrigger>
   394→   394→                  <SelectContent>
   395→   395→                    {lists.map((l) => <SelectItem key={l.id} value={l.id}>{l.name}</SelectItem>)}
   396→   396→                  </SelectContent>
   397→   397→                </Select>
   398→   398→              </div>
   399→   399→            )}
   400→   400→
   401→   401→            {channel === "email" && (
   402→   402→              <>
   403→   403→                <div className="space-y-1.5">
   404→   404→                  <div className="flex items-center justify-between">

   514→   514→function SendDialog({
   515→   515→  campaign, onClose, onQueued,
   516→   516→}: {
   517→   517→  campaign: Campaign; onClose: () => void; onQueued: () => void;
   518→   518→}) {
   519→   519→  const { toast } = useToast();
   520→   520→  const [mode, setMode] = useState<"now" | "schedule">("now");
   521→   521→  const [when, setWhen] = useState("");
   522→   522→  const [testTo, setTestTo] = useState("");
   523→   523→  const [busy, setBusy] = useState(false);
   524→   524→  const [testBusy, setTestBusy] = useState(false);
   525→   525→
   526→   526→  async function send() {
   527→   527→    setBusy(true);
   528→   528→    try {
   529→   529→      await api.post(`/api/campaigns/${campaign.id}/send`, {
   530→   530→        mode,
   531→   531→        scheduledAt: mode === "schedule" && when ? new Date(when).toISOString() : undefined,
   532→   532→      });
   533→   533→      onQueued();
   534→   534→    } catch {
   535→   535→      toast({ title: "Send failed", description: "The campaign could not be queued.", variant: "destructive" });
   536→   536→    } finally {
   537→   537→      setBusy(false);
   538→   538→    }
   539→   539→  }
   540→   540→
   541→   541→  async function sendTest() {
   542→   542→    if (!testTo.trim()) return;
   543→   543→    setTestBusy(true);
   544→   544→    try {
   545→   545→      await api.post(`/api/campaigns/${campaign.id}/test`, {
   546→   546→        channel: campaign.channel,
   547→   547→        to: campaign.channel === "email" ? testTo : testTo,
   548→   548→      });
   549→   549→      toast({ title: "Test queued", description: `Sending a test to ${testTo}` });
   550→   550→    } catch {
   551→   551→      toast({ title: "Test send failed", variant: "destructive" });
   552→   552→    } finally {
   553→   553→      setTestBusy(false);
   554→   554→    }
   555→   555→  }
   556→   556→
   557→   557→  return (
   558→   558→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   559→   559→      <DialogContent className="max-w-md">
   560→   560→        <DialogHeader>
   561→   561→          <DialogTitle>Send &quot;{campaign.name}&quot;</DialogTitle>
   562→   562→          <DialogDescription>
   563→   563→            The automation engine dispatches with per-message evidence and retry protection.
   564→   564→          </DialogDescription>
   565→   565→        </DialogHeader>
   566→   566→
   567→   567→        <div className="space-y-4">
   568→   568→          <div className="grid grid-cols-2 gap-2">
   569→   569→            <Button
   570→   570→              variant={mode === "now" ? "default" : "outline"}
   571→   571→              onClick={() => setMode("now")}
   572→   572→              className={mode === "now" ? "bg-teal-600 hover:bg-teal-700" : ""}
   573→   573→            >
   574→   574→              <Send className="mr-1.5 h-4 w-4" aria-hidden /> Send now
   575→   575→            </Button>
   576→   576→            <Button variant={mode === "schedule" ? "default" : "outline"} onClick={() => setMode("schedule")} className={mode === "schedule" ? "bg-teal-600 hover:bg-teal-700" : ""}>
   577→   577→              <Clock className="mr-1.5 h-4 w-4" aria-hidden /> Schedule
   578→   578→            </Button>
   579→   579→          </div>
   580→   580→
   581→   581→          {mode === "schedule" && (
   582→   582→            <div className="space-y-1.5">
   583→   583→              <Label htmlFor="send-when">Date & time</Label>
   584→   584→              <Input id="send-when" type="datetime-local" value={when} onChange={(e) => setWhen(e.target.value)} />
   585→   585→            </div>
   586→   586→          )}
   587→   587→
   588→   588→          <div className="rounded-lg border border-dashed p-3">
   589→   589→            <p className="mb-2 flex items-center gap-1.5 text-xs font-medium text-muted-foreground">
   590→   590→              <FlaskConical className="h-3.5 w-3.5" aria-hidden /> Send a test first
   591→   591→            </p>
   592→   592→            <div className="flex gap-2">
   593→   593→              <Input
   594→   594→                value={testTo}
   595→   595→                onChange={(e) => setTestTo(e.target.value)}
   596→   596→                placeholder={campaign.channel === "email" ? "you@company.com" : "+92300…"}
   597→   597→                aria-label="Test recipient"
   598→   598→              />
   599→   599→              <Button variant="secondary" size="sm" onClick={sendTest} disabled={testBusy || !testTo.trim()}>
   600→   600→                {testBusy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : "Send"}
   601→   601→              </Button>
   602→   602→            </div>
   603→   603→          </div>
   604→   604→        </div>
   605→   605→
   606→   606→        <DialogFooter>
   607→   607→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   608→   608→          <Button onClick={send} disabled={busy || (mode === "schedule" && !when)} className="bg-teal-600 hover:bg-teal-700">
   609→   609→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   610→   610→            {mode === "now" ? "Send to audience" : "Schedule campaign"}
   611→   611→          </Button>
   612→   612→        </DialogFooter>
   613→   613→      </DialogContent>
   614→   614→    </Dialog>
   615→   615→  );
