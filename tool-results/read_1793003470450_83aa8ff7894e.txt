   231→// ─── List editor ─────────────────────────────────────────────
   232→
   233→function ListEditorDialog({
   234→  open, list, onOpenChange, onSaved,
   235→}: {
   236→  open: boolean; list: ListRow | null;
   237→  onOpenChange: (o: boolean) => void; onSaved: (isNew: boolean) => void;
   238→}) {
   239→  const { toast } = useToast();
   240→  const [name, setName] = useState("");
   241→  const [description, setDescription] = useState("");
   242→  const [busy, setBusy] = useState(false);
   243→
   244→  useMemo(() => {
   245→    if (list) {
   246→      setName(list.name);
   247→      setDescription(list.description ?? "");
   248→    } else {
   249→      setName(""); setDescription("");
   250→    }
   251→  }, [list, open]);
   252→
   253→  async function save() {
   254→    if (!name.trim()) {
   255→      toast({ title: "Name required", variant: "destructive" });
   256→      return;
   257→    }
   258→    setBusy(true);
   259→    try {
   260→      const payload = { name: name.trim(), description: description.trim().slice(0, 500) };
   261→      if (list) await api.patch(`/api/lists/${list.id}`, payload);
   262→      else await api.post("/api/lists", payload);
   263→      onSaved(!list);
   264→    } catch (err) {
   265→      toast({ title: "Could not save list", description: describeError(err), variant: "destructive" });
   266→    } finally {
   267→      setBusy(false);
   268→    }
   269→  }
   270→
   271→  return (
   272→    <Dialog open={open} onOpenChange={onOpenChange}>
   273→      <DialogContent className="sm:max-w-md">
   274→        <DialogHeader>
   275→          <DialogTitle>{list ? "Edit list" : "New list"}</DialogTitle>
   276→          <DialogDescription>Lists are static segments of contacts you curate manually.</DialogDescription>
   277→        </DialogHeader>
   278→        <div className="space-y-3">
   279→          <div className="space-y-1.5">
   280→            <Label htmlFor="list-name">Name *</Label>
   281→            <Input
   282→              id="list-name"
   283→              value={name}
   284→              onChange={(e) => setName(e.target.value)}
   285→              placeholder="Newsletter — weekly"
   286→              maxLength={120}
   287→              autoFocus
   288→            />
   289→          </div>
   290→          <div className="space-y-1.5">
   291→            <Label htmlFor="list-desc">Description</Label>
   292→            <Textarea
   293→              id="list-desc"
   294→              value={description}
   295→              onChange={(e) => setDescription(e.target.value)}
   296→              placeholder="What is this list for?"
   297→              maxLength={500}
   298→              className="min-h-[80px]"
   299→            />
   300→          </div>
   301→        </div>
   302→        <DialogFooter>
   303→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>Cancel</Button>
   304→          <Button onClick={save} disabled={busy} className="bg-teal-600 hover:bg-teal-700">
   305→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   306→            {list ? "Save changes" : "Create list"}
   307→          </Button>
   308→        </DialogFooter>
   309→      </DialogContent>
   310→    </Dialog>
   311→  );
   312→}

   506→// ─── Segment rule builder ────────────────────────────────────
   507→
   508→interface EditableRule { field: string; operator: string; value: string; conjunction: "AND" | "OR" }
   509→
   510→function SegmentEditorDialog({
   511→  open, segment, onOpenChange, onSaved,
   512→}: {
   513→  open: boolean; segment: Segment | null;
   514→  onOpenChange: (o: boolean) => void; onSaved: (isNew: boolean) => void;
   515→}) {
   516→  const { toast } = useToast();
   517→  const [name, setName] = useState("");
   518→  const [description, setDescription] = useState("");
   519→  const [rules, setRules] = useState<EditableRule[]>([]);
   520→  const [preview, setPreview] = useState<SegmentPreview | null>(null);
   521→  const [busy, setBusy] = useState(false);
   522→
   523→  useMemo(() => {
   524→    if (segment) {
   525→      setName(segment.name);
   526→      setDescription(segment.description ?? "");
   527→      setRules(
   528→        segment.rules.length > 0
   529→          ? segment.rules.map((r) => ({ field: r.field, operator: r.operator, value: r.value ?? "", conjunction: r.conjunction ?? "AND" }))
   530→          : [newRule()]
   531→      );
   532→    } else {
   533→      setName(""); setDescription(""); setRules([newRule()]);
   534→    }
   535→    setPreview(null);
   536→  }, [segment, open]);
   537→
   538→  function newRule(): EditableRule {
   539→    return { field: "email", operator: "contains", value: "", conjunction: "AND" };
   540→  }
   541→
   542→  function updateRule(index: number, patch: Partial<EditableRule>) {
   543→    setRules((prev) => prev.map((r, i) => (i === index ? { ...r, ...patch } : r)));
   544→  }
   545→
   546→  function removeRule(index: number) {
   547→    setRules((prev) => prev.filter((_, i) => i !== index));
   548→  }
   549→
   550→  const rulesPayload = useMemo(
   551→    () =>
   552→      rules.map((r) => ({
   553→        field: r.field,
   554→        operator: r.operator,
   555→        value: NO_VALUE_OPERATORS.has(r.operator) ? "" : r.value.trim(),
   556→        conjunction: r.conjunction,
   557→      })),
   558→    [rules]
   559→  );
   560→
   561→  async function runPreview() {
   562→    setBusy(true);
   563→    try {
   564→      const data = await api.put<SegmentPreview>("/api/segments", {
   565→        name: name.trim() || "Preview",
   566→        description: description.trim(),
   567→        rules: rulesPayload,
   568→      });
   569→      setPreview(data);
   570→    } catch (err) {
   571→      toast({ title: "Preview failed", description: describeError(err), variant: "destructive" });
   572→    } finally {
   573→      setBusy(false);
   574→    }
   575→  }
   576→
   577→  async function save() {
   578→    if (!name.trim()) {
   579→      toast({ title: "Name required", variant: "destructive" });
   580→      return;
   581→    }
   582→    if (rules.length === 0) {
   583→      toast({ title: "Add at least one condition", variant: "destructive" });
   584→      return;
   585→    }
   586→    setBusy(true);
   587→    try {
   588→      const payload = { name: name.trim(), description: description.trim(), rules: rulesPayload };
   589→      if (segment) await api.patch(`/api/segments/${segment.id}`, payload);
   590→      else await api.post("/api/segments", payload);
   591→      onSaved(!segment);
   592→    } catch (err) {
   593→      toast({ title: "Could not save segment", description: describeError(err), variant: "destructive" });
   594→    } finally {
   595→      setBusy(false);
   596→    }
   597→  }
   598→
   599→  return (
   600→    <Dialog open={open} onOpenChange={onOpenChange}>
   601→      <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-2xl">
   602→        <DialogHeader>
   603→          <DialogTitle>{segment ? "Edit segment" : "New segment"}</DialogTitle>
   604→          <DialogDescription>
   605→            Combine conditions with AND / OR. Use “Preview match” to see how many contacts qualify before saving.
   606→          </DialogDescription>
   607→        </DialogHeader>
   608→
   609→        <div className="grid gap-3 sm:grid-cols-2">
   610→          <div className="space-y-1.5">
   611→            <Label htmlFor="seg-name">Name *</Label>
   612→            <Input
   613→              id="seg-name"
   614→              value={name}
   615→              onChange={(e) => setName(e.target.value)}
   616→              placeholder="Hot leads in the US"
   617→              maxLength={120}
   618→            />
   619→          </div>
   620→          <div className="space-y-1.5">
   621→            <Label htmlFor="seg-desc">Description</Label>
   622→            <Input
   623→              id="seg-desc"
   624→              value={description}
   625→              onChange={(e) => setDescription(e.target.value)}
   626→              placeholder="What does this segment target?"
   627→              maxLength={500}
   628→            />
   629→          </div>
   630→        </div>
   631→
   632→        <div className="space-y-2">
   633→          <Label>Conditions</Label>
   634→          {rules.map((rule, i) => {
   635→            const needsValue = !NO_VALUE_OPERATORS.has(rule.operator);
   636→            return (
   637→              <div key={i} className="flex flex-wrap items-center gap-2 rounded-lg border bg-muted/30 p-2">
   638→                {i > 0 ? (
   639→                  <Select
   640→                    value={rule.conjunction}
   641→                    onValueChange={(v) => updateRule(i, { conjunction: v as "AND" | "OR" })}
   642→                  >
   643→                    <SelectTrigger className="h-9 w-[74px]" aria-label={`Conjunction for condition ${i + 1}`}>
   644→                      <SelectValue />
   645→                    </SelectTrigger>
   646→                    <SelectContent>
   647→                      <SelectItem value="AND">AND</SelectItem>
   648→                      <SelectItem value="OR">OR</SelectItem>
   649→                    </SelectContent>
   650→                  </Select>
   651→                ) : (
   652→                  <span className="inline-flex h-9 min-w-[74px] items-center justify-center rounded-md border border-dashed px-2 text-xs font-medium text-muted-foreground">
   653→                    WHERE
   654→                  </span>
   655→                )}
   656→
   657→                <Select value={rule.field} onValueChange={(v) => updateRule(i, { field: v })}>
   658→                  <SelectTrigger className="h-9 w-[132px]" aria-label={`Field for condition ${i + 1}`}>
   659→                    <SelectValue />
   660→                  </SelectTrigger>
   661→                  <SelectContent>
   662→                    {SEGMENT_FIELDS.map((f) => (
   663→                      <SelectItem key={f} value={f}>{humanize(f)}</SelectItem>
   664→                    ))}
   665→                  </SelectContent>
   666→                </Select>
   667→
   668→                <Select value={rule.operator} onValueChange={(v) => updateRule(i, { operator: v })}>
   669→                  <SelectTrigger className="h-9 w-[150px]" aria-label={`Operator for condition ${i + 1}`}>
   670→                    <SelectValue />
   671→                  </SelectTrigger>
   672→                  <SelectContent>
   673→                    {SEGMENT_OPERATORS.map((op) => (
   674→                      <SelectItem key={op} value={op}>{humanize(op)}</SelectItem>
   675→                    ))}
   676→                  </SelectContent>
   677→                </Select>
   678→
   679→                {needsValue ? (
   680→                  <Input
   681→                    value={rule.value}
   682→                    onChange={(e) => updateRule(i, { value: e.target.value })}
   683→                    placeholder={rule.operator === "in" ? "a, b, c" : "Value"}
   684→                    className="h-9 min-w-[110px] flex-1"
   685→                    maxLength={200}
   686→                    aria-label={`Value for condition ${i + 1}`}
   687→                  />
   688→                ) : (
   689→                  <span className="min-w-[110px] flex-1 text-xs italic text-muted-foreground">no value needed</span>
   690→                )}
   691→
   692→                <Button
   693→                  variant="ghost"
   694→                  size="icon"
   695→                  className="h-9 w-9 shrink-0 text-muted-foreground hover:text-red-600"
   696→                  onClick={() => removeRule(i)}
   697→                  disabled={rules.length === 1}
   698→                  aria-label={`Remove condition ${i + 1}`}
   699→                >
   700→                  <X className="h-4 w-4" aria-hidden />
   701→                </Button>
   702→              </div>
   703→            );
   704→          })}
   705→
   706→          <div className="flex flex-wrap items-center gap-2">
   707→            <Button
   708→              variant="outline"
   709→              size="sm"
   710→              onClick={() => setRules((prev) => [...prev, newRule()])}
   711→              disabled={rules.length >= 20}
   712→            >
   713→              <Plus className="mr-1 h-3.5 w-3.5" aria-hidden /> Add condition
   714→            </Button>
   715→            <Button
   716→              variant="outline"
   717→              size="sm"
   718→              onClick={runPreview}
   719→              disabled={busy || rules.length === 0}
   720→              className="border-teal-200 text-teal-700 hover:bg-teal-50 hover:text-teal-800 dark:border-teal-900 dark:text-teal-400 dark:hover:bg-teal-950"
   721→            >
   722→              {busy ? <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin" aria-hidden /> : <FlaskConical className="mr-1 h-3.5 w-3.5" aria-hidden />}
   723→              Preview match
   724→            </Button>
   725→          </div>
   726→
   727→          {preview && (
   728→            <div className="rounded-lg border bg-teal-50/60 p-3 dark:border-teal-900 dark:bg-teal-950/30">
   729→              <p className="text-sm font-medium text-teal-800 dark:text-teal-300">
   730→                {formatNumber(preview.matchedCount)} of {formatNumber(preview.total)} contacts match
   731→              </p>
   732→              {preview.sample.length > 0 && (
   733→                <div className="mt-2 flex flex-wrap gap-1">
   734→                  {preview.sample.map((s) => (
   735→                    <Badge key={s.id} variant="secondary" className="max-w-[220px]">
   736→                      <span className="truncate">{s.email}</span>
   737→                    </Badge>
   738→                  ))}
   739→                </div>
   740→              )}
   741→            </div>
   742→          )}
   743→        </div>
   744→
   745→        <DialogFooter>
   746→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>Cancel</Button>
   747→          <Button onClick={save} disabled={busy} className="bg-teal-600 hover:bg-teal-700">
   748→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   749→            {segment ? "Save changes" : "Create segment"}
   750→          </Button>
   751→        </DialogFooter>
   752→      </DialogContent>
   753→    </Dialog>
   754→  );
   755→}
