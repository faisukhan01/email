    76→    76→const HEADER_ALIASES: Record<string, string> = {

    86→    86→function splitCsvLine(line: string): string[] {
    87→    87→  const cells: string[] = [];
    88→    88→  let cur = "";
    89→    89→  let inQuotes = false;
    90→    90→  for (let i = 0; i < line.length; i++) {
    91→    91→    const ch = line[i];
    92→    92→    if (inQuotes) {
    93→    93→      if (ch === '"') {
    94→    94→        if (line[i + 1] === '"') { cur += '"'; i++; } else { inQuotes = false; }
    95→    95→      } else {
    96→    96→        cur += ch;
    97→    97→      }
    98→    98→    } else if (ch === '"') {
    99→    99→      inQuotes = true;
   100→   100→    } else if (ch === ",") {
   101→   101→      cells.push(cur); cur = "";
   102→   102→    } else {
   103→   103→      cur += ch;
   104→   104→    }
   105→   105→  }
   106→   106→  cells.push(cur);
   107→   107→  return cells.map((c) => c.trim());
   108→   108→}
   109→   109→
   110→   110→function parseCsv(text: string): { rows: Record<string, string>[]; error: string | null } {
   111→   111→  const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean);
   112→   112→  if (lines.length < 2) return { rows: [], error: "Paste at least a header row and one data row." };
   113→   113→  const headers = splitCsvLine(lines[0]).map((h) => HEADER_ALIASES[h.toLowerCase().replace(/[^a-z0-9]/g, "")] ?? "");
   114→   114→  if (!headers.includes("email")) {
   115→   115→    return { rows: [], error: "No email column found. Include a header named 'email'." };
   116→   116→  }
   117→   117→  const rows: Record<string, string>[] = [];
   118→   118→  for (const line of lines.slice(1)) {
   119→   119→    const cells = splitCsvLine(line);
   120→   120→    const row: Record<string, string> = {};
   121→   121→    headers.forEach((h, i) => {
   122→   122→      if (h && FORM_FIELDS.includes(h as (typeof FORM_FIELDS)[number])) row[h] = cells[i] ?? "";
   123→   123→    });
   124→   124→    if (row.email) rows.push(row);
   125→   125→  }
   126→   126→  return { rows, error: rows.length === 0 ? "No data rows with an email found." : null };
   127→   127→}

   722→   722→function TagDialog({
   723→   723→  open, mode, contact, onOpenChange,
   724→   724→}: {
   725→   725→  open: boolean; mode: "add" | "remove"; contact: Contact | null;
   726→   726→  onOpenChange: (o: boolean) => void;
   727→   727→}) {
   728→   728→  const qc = useQueryClient();
   729→   729→  const { toast } = useToast();
   730→   730→  const [tag, setTag] = useState("");
   731→   731→
   732→   732→  const patchTags = useMutation({
   733→   733→    mutationFn: (vars: { id: string; tags: string[]; verb: string }) =>
   734→   734→      api.patch(`/api/contacts/${vars.id}`, { tags: vars.tags }).then(() => vars.verb),
   735→   735→    onSuccess: (verb) => {
   736→   736→      qc.invalidateQueries({ queryKey: ["contacts"] });
   737→   737→      qc.invalidateQueries({ queryKey: ["contact"] });
   738→   738→      toast({ title: verb === "added" ? "Tag added" : "Tag removed" });
   739→   739→    },
   740→   740→    onError: (err) => toast({ title: "Could not update tags", description: describeError(err), variant: "destructive" }),
   741→   741→  });
   742→   742→
   743→   743→  function addTag() {
   744→   744→    const t = tag.trim().slice(0, 40);
   745→   745→    if (!t || !contact) return;
   746→   746→    if (contact.tags.includes(t)) {
   747→   747→      toast({ title: "Tag already exists on this contact", variant: "destructive" });
   748→   748→      return;
   749→   749→    }
   750→   750→    patchTags.mutate({ id: contact.id, tags: [...contact.tags, t].slice(0, 30), verb: "added" });
   751→   751→    setTag("");
   752→   752→  }
   753→   753→
   754→   754→  function removeTag(t: string) {
   755→   755→    if (!contact) return;
   756→   756→    patchTags.mutate({ id: contact.id, tags: contact.tags.filter((x) => x !== t), verb: "removed" });
   757→   757→  }
   758→   758→
   759→   759→  return (
   760→   760→    <Dialog open={open} onOpenChange={(o) => { if (!o) setTag(""); onOpenChange(o); }}>
   761→   761→      <DialogContent className="sm:max-w-sm">
   762→   762→        <DialogHeader>
   763→   763→          <DialogTitle>{mode === "add" ? "Add tag" : "Remove tag"}</DialogTitle>
   764→   764→          <DialogDescription className="truncate">
   765→   765→            {contact ? contactName(contact) : ""}
   766→   766→          </DialogDescription>
   767→   767→        </DialogHeader>
   768→   768→
   769→   769→        {mode === "add" ? (
   770→   770→          <div className="space-y-3">
   771→   771→            <div className="space-y-1.5">
   772→   772→              <Label htmlFor="row-tag">Tag name</Label>
   773→   773→              <Input
   774→   774→                id="row-tag"
   775→   775→                value={tag}
   776→   776→                onChange={(e) => setTag(e.target.value)}
   777→   777→                onKeyDown={(e) => { if (e.key === "Enter") { e.preventDefault(); addTag(); } }}
   778→   778→                placeholder="vip"
   779→   779→                maxLength={40}
   780→   780→                autoFocus
   781→   781→              />
   782→   782→            </div>
   783→   783→            {contact && contact.tags.length > 0 && (
   784→   784→              <div className="flex flex-wrap gap-1">
   785→   785→                {contact.tags.map((t) => (
   786→   786→                  <Badge key={t} variant="secondary">{t}</Badge>
   787→   787→                ))}
   788→   788→              </div>
   789→   789→            )}
   790→   790→          </div>
   791→   791→        ) : (
   792→   792→          <div className="flex flex-wrap gap-1.5">
   793→   793→            {contact?.tags.map((t) => (
   794→   794→              <Badge key={t} variant="secondary" className="gap-1 py-1 pl-2.5">
   795→   795→                {t}
   796→   796→                <button
   797→   797→                  onClick={() => removeTag(t)}
   798→   798→                  className="rounded-full p-0.5 hover:bg-stone-200 dark:hover:bg-stone-700"
   799→   799→                  aria-label={`Remove tag ${t}`}
   800→   800→                  disabled={patchTags.isPending}
   801→   801→                >
   802→   802→                  <X className="h-3 w-3" aria-hidden />
   803→   803→                </button>
   804→   804→              </Badge>
   805→   805→            ))}
   806→   806→          </div>
   807→   807→        )}
   808→   808→
   809→   809→        <DialogFooter>
   810→   810→          {mode === "add" ? (
   811→   811→            <Button onClick={addTag} disabled={patchTags.isPending || !tag.trim()} className="bg-teal-600 hover:bg-teal-700">
   812→   812→              {patchTags.isPending && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   813→   813→              Add tag
   814→   814→            </Button>
   815→   815→          ) : (
   816→   816→            <Button variant="outline" onClick={() => onOpenChange(false)}>Done</Button>
   817→   817→          )}
   818→   818→        </DialogFooter>
   819→   819→      </DialogContent>
   820→   820→    </Dialog>
   821→   821→  );

   874→   874→function ImportDialog({
   875→   875→  open, lists, onOpenChange,
   876→   876→}: {
   877→   877→  open: boolean; lists: ListRow[]; onOpenChange: (o: boolean) => void;
   878→   878→}) {
   879→   879→  const qc = useQueryClient();
   880→   880→  const { toast } = useToast();
   881→   881→  const [csvText, setCsvText] = useState("");
   882→   882→  const [listId, setListId] = useState("none");
   883→   883→  const [result, setResult] = useState<ImportResult | null>(null);
   884→   884→
   885→   885→  const parsed = useMemo(() => parseCsv(csvText), [csvText]);
   886→   886→
   887→   887→  const importMutation = useMutation({
   888→   888→    mutationFn: (vars: { contacts: Record<string, string>[]; listId?: string }) =>
   889→   889→      api.post<ImportResult>("/api/contacts/import", {
   890→   890→        contacts: vars.contacts.slice(0, 2000),
   891→   891→        ...(vars.listId ? { listId: vars.listId } : {}),
   892→   892→      }),
   893→   893→    onSuccess: (data) => {
   894→   894→      setResult(data);
   895→   895→      qc.invalidateQueries({ queryKey: ["contacts"] });
   896→   896→      qc.invalidateQueries({ queryKey: ["lists"] });
   897→   897→      toast({
   898→   898→        title: "Import finished",
   899→   899→        description: `${data.created} created · ${data.updated} updated · ${data.skipped} skipped.`,
   900→   900→      });
   901→   901→    },
   902→   902→    onError: (err) => toast({ title: "Import failed", description: describeError(err), variant: "destructive" }),
   903→   903→  });
   904→   904→
   905→   905→  function reset() {
   906→   906→    setCsvText(""); setListId("none"); setResult(null);
   907→   907→  }
   908→   908→
   909→   909→  function submit() {
   910→   910→    if (parsed.error || parsed.rows.length === 0) {
   911→   911→      toast({ title: "Cannot import", description: parsed.error ?? "Nothing to import.", variant: "destructive" });
   912→   912→      return;
   913→   913→    }
   914→   914→    importMutation.mutate({ contacts: parsed.rows, listId: listId === "none" ? undefined : listId });
   915→   915→  }
   916→   916→
   917→   917→  return (
   918→   918→    <Dialog open={open} onOpenChange={(o) => { onOpenChange(o); if (!o) reset(); }}>
   919→   919→      <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-xl">
   920→   920→        <DialogHeader>
   921→   921→          <DialogTitle>Import contacts from CSV</DialogTitle>
   922→   922→          <DialogDescription>
   923→   923→            Paste CSV text or upload a file. The first row must be headers — recognized columns: email, firstName, lastName, phone, company, country, city, jobTitle.
   924→   924→          </DialogDescription>
   925→   925→        </DialogHeader>
   926→   926→
   927→   927→        {result ? (
   928→   928→          <div className="space-y-3">
   929→   929→            <div className="grid grid-cols-3 gap-2">
   930→   930→              <div className="rounded-lg border p-3 text-center">
   931→   931→                <p className="text-xl font-semibold text-teal-700 dark:text-teal-400">{result.created}</p>
   932→   932→                <p className="text-xs text-muted-foreground">Created</p>
   933→   933→              </div>
   934→   934→              <div className="rounded-lg border p-3 text-center">
   935→   935→                <p className="text-xl font-semibold text-amber-700 dark:text-amber-400">{result.updated}</p>
   936→   936→                <p className="text-xs text-muted-foreground">Updated</p>
   937→   937→              </div>
   938→   938→              <div className="rounded-lg border p-3 text-center">
   939→   939→                <p className="text-xl font-semibold text-muted-foreground">{result.skipped}</p>
   940→   940→                <p className="text-xs text-muted-foreground">Skipped</p>
   941→   941→              </div>
   942→   942→            </div>
   943→   943→            {result.errors.length > 0 && (
   944→   944→              <div className="rounded-lg border border-orange-200 bg-orange-50 p-3 dark:border-orange-900 dark:bg-orange-950/40">
   945→   945→                <p className="flex items-center gap-1.5 text-sm font-medium text-orange-800 dark:text-orange-300">
   946→   946→                  <FileWarning className="h-4 w-4" aria-hidden /> {result.errors.length} row{result.errors.length === 1 ? "" : "s"} skipped
   947→   947→                </p>
   948→   948→                <ul className="mt-2 space-y-1 text-xs text-orange-800/90 dark:text-orange-300/90">
   949→   949→                  {result.errors.slice(0, 5).map((e, i) => (
   950→   950→                    <li key={i}>Row {e.row} ({e.email}): {e.reason}</li>
   951→   951→                  ))}
   952→   952→                  {result.errors.length > 5 && <li>…and {result.errors.length - 5} more</li>}
   953→   953→                </ul>
   954→   954→              </div>

   956→   956→          </div>
   957→   957→        ) : (

   969→   969→                    const text = await file.text();
   970→   970→                    setCsvText(text);
   971→   971→                    setResult(null);
   972→   972→                    toast({ title: `Loaded ${file.name}`, description: "Review the parsed row count below." });
   973→   973→                  } catch {
   974→   974→                    toast({ title: "Could not read file", variant: "destructive" });
   975→   975→                  }
   976→   976→                }}
   977→   977→              />
   978→   978→            </div>

   997→   997→                  : "Waiting for CSV input…"}
