    57→function StatBar({ stats }: { stats: Record<string, number> }) {
    58→  const total = Object.values(stats).reduce((a, b) => a + b, 0);
    59→  if (total === 0) return <span className="text-xs text-muted-foreground">Not sent yet</span>;
    60→  const clicked = stats.clicked ?? 0;
    61→  const opened = (stats.opened ?? 0) + clicked;
    62→  const sent = total - (stats.bounced ?? 0) - (stats.failed ?? 0) - (stats.queued ?? 0);
    63→  const pct = (n: number) => (sent > 0 ? Math.round((n / sent) * 100) : 0);
    64→  return (
    65→    <div className="space-y-1">
    66→      <div className="flex gap-0.5">
    67→        <div className="h-1.5 rounded-l bg-teal-500" style={{ width: `${Math.max(pct(sent - opened), opened < sent ? 8 : 4)}%` }} />
    68→        <div className="h-1.5 bg-amber-400" style={{ width: `${Math.max(pct(opened - clicked), opened > 0 ? 6 : 0)}%` }} />
    69→        <div className="h-1.5 rounded-r bg-lime-500" style={{ width: `${Math.max(pct(clicked), clicked > 0 ? 6 : 0)}%` }} />
    70→      </div>
    71→      <p className="text-[11px] text-muted-foreground">
    72→        {formatNumber(total)} recipients · {pct(opened)}% opens · {pct(clicked)}% clicks
    73→      </p>
    74→    </div>
    75→  );
    76→}

   277→function CampaignEditor({
   278→  open, campaign, lists, onOpenChange, onSaved, previewMode, setPreviewMode,
   279→}: {
   280→  open: boolean; campaign: Campaign | null; lists: { id: string; name: string }[];
   281→  onOpenChange: (v: boolean) => void; onSaved: () => void;
   282→  previewMode: boolean; setPreviewMode: (v: boolean) => void;
   283→}) {
   284→  const { toast } = useToast();
   285→  const [name, setName] = useState("");
   286→  const [channel, setChannel] = useState("email");
   287→  const [subject, setSubject] = useState("");
   288→  const [preheader, setPreheader] = useState("");
   289→  const [fromName, setFromName] = useState("");
   290→  const [fromEmail, setFromEmail] = useState("");
   291→  const [bodyHtml, setBodyHtml] = useState(DEFAULT_HTML);
   292→  const [bodyText, setBodyText] = useState("");
   293→  const [audienceType, setAudienceType] = useState("list");
   294→  const [listId, setListId] = useState("");
   295→  const [busy, setBusy] = useState(false);
   296→
   297→  useMemo(() => {
   298→    if (campaign) {
   299→      setName(campaign.name); setChannel(campaign.channel); setSubject(campaign.subject);
   300→      setPreheader(campaign.preheader); setFromName(campaign.fromName); setFromEmail(campaign.fromEmail);
   301→      setBodyHtml(campaign.bodyHtml || DEFAULT_HTML); setBodyText(campaign.bodyText);
   302→      setAudienceType(campaign.audienceType); setListId(campaign.listId ?? "");
   303→    } else {
   304→      setName(""); setChannel("email"); setSubject(""); setPreheader("");
   305→      setFromName(""); setFromEmail(""); setBodyHtml(DEFAULT_HTML); setBodyText("");
   306→      setAudienceType("list"); setListId("");
   307→    }
   308→  }, [campaign, open]);
   309→
   310→  const previewHtml = useMemo(
   311→    () =>
   312→      (bodyHtml || "<p></p>")
   313→        .replace(/\{\{\s*first_name\s*\}\}/g, "Alex")
   314→        .replace(/\{\{\s*last_name\s*\}\}/g, "Morgan")
   315→        .replace(/\{\{\s*full_name\s*\}\}/g, "Alex Morgan")
   316→        .replace(/\{\{\s*company\s*\}\}/g, "Acme Inc")
   317→        .replace(/\{\{\s*(\w+)\s*\}\}/g, "—"),
   318→    [bodyHtml]
   319→  );
   320→
   321→  async function save() {
   322→    if (!name.trim()) {
   323→      toast({ title: "Name required", variant: "destructive" });
   324→      return;
   325→    }
   326→    setBusy(true);
   327→    try {
   328→      const payload = {
   329→        name, channel, subject, preheader, fromName, fromEmail,
   330→        bodyHtml: channel === "email" ? bodyHtml : "",
   331→        bodyText: channel === "whatsapp" ? bodyText : "",
   332→        audienceType,
   333→        listId: audienceType === "list" && listId ? listId : undefined,
   334→      };
   335→      if (campaign) {
   336→        await api.patch(`/api/campaigns/${campaign.id}`, payload);
   337→      } else {
   338→        await api.post("/api/campaigns", payload);
   339→      }
   340→      onSaved();
   341→    } catch {
   342→      toast({ title: "Could not save campaign", variant: "destructive" });
   343→    } finally {
   344→      setBusy(false);
   345→    }
   346→  }
   347→
   348→  return (
   349→    <Dialog open={open} onOpenChange={onOpenChange}>
   350→      <DialogContent className="max-w-3xl">
   351→        <DialogHeader>
   352→          <DialogTitle>{campaign ? "Edit campaign" : "New campaign"}</DialogTitle>
   353→          <DialogDescription>
   354→            Personalize with tokens — the engine renders them per contact at send time.
   355→          </DialogDescription>
   356→        </DialogHeader>
   357→
   358→        <div className="grid gap-4 sm:grid-cols-2">
   359→          <div className="space-y-3">
   360→            <div className="space-y-1.5">
   361→              <Label htmlFor="camp-name">Campaign name</Label>
   362→              <Input id="camp-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="October Newsletter" maxLength={140} />
   363→            </div>
   364→
   365→            <div className="grid grid-cols-2 gap-3">
   366→              <div className="space-y-1.5">
   367→                <Label>Channel</Label>
   368→                <Select value={channel} onValueChange={setChannel}>
   369→                  <SelectTrigger aria-label="Channel"><SelectValue /></SelectTrigger>
   370→                  <SelectContent>
   371→                    <SelectItem value="email">Email</SelectItem>
   372→                    <SelectItem value="whatsapp">WhatsApp</SelectItem>
   373→                  </SelectContent>
   374→                </Select>
   375→              </div>
   376→              <div className="space-y-1.5">
   377→                <Label>Audience</Label>
   378→                <Select value={audienceType} onValueChange={setAudienceType}>
   379→                  <SelectTrigger aria-label="Audience"><SelectValue /></SelectTrigger>
   380→                  <SelectContent>
   381→                    <SelectItem value="list">A list</SelectItem>
   382→                    <SelectItem value="all">All contacts</SelectItem>
   383→                    <SelectItem value="segment">Segment rules</SelectItem>
   384→                  </SelectContent>
   385→                </Select>
   386→              </div>
   387→            </div>
   388→
   389→            {audienceType === "list" && (
   390→              <div className="space-y-1.5">
   391→                <Label>Send to list</Label>
   392→                <Select value={listId} onValueChange={setListId}>
   393→                  <SelectTrigger aria-label="List"><SelectValue placeholder="Choose a list" /></SelectTrigger>
   394→                  <SelectContent>
   395→                    {lists.map((l) => <SelectItem key={l.id} value={l.id}>{l.name}</SelectItem>)}
   396→                  </SelectContent>
   397→                </Select>
   398→              </div>
   399→            )}
   400→
   401→            {channel === "email" && (
   402→              <>
   403→                <div className="space-y-1.5">
   404→                  <div className="flex items-center justify-between">
   405→                    <Label htmlFor="camp-subject">Subject line</Label>
   406→                    <div className="flex gap-1">
   407→                      {TOKENS.slice(0, 3).map((t) => (
   408→                        <button
   409→                          key={t}
   410→                          type="button"
   411→                          className="rounded border px-1.5 py-0.5 text-[10px] text-muted-foreground hover:bg-muted"
   412→                          onClick={() => setSubject((s) => `${s}{{${t}}}`)}
   413→                        >
   414→                          +{t}
   415→                        </button>
   416→                      ))}
   417→                    </div>
   418→                  </div>
   419→                  <Input id="camp-subject" value={subject} onChange={(e) => setSubject(e.target.value)} placeholder="Hey {{first_name}}, quick question…" maxLength={200} />
   420→                </div>
   421→                <div className="space-y-1.5">
   422→                  <Label htmlFor="camp-preheader">Preheader</Label>
   423→                  <Input id="camp-preheader" value={preheader} onChange={(e) => setPreheader(e.target.value)} placeholder="Preview text shown in the inbox" maxLength={200} />
   424→                </div>
   425→                <div className="grid grid-cols-2 gap-3">
   426→                  <div className="space-y-1.5">
   427→                    <Label htmlFor="camp-fromname">From name</Label>
   428→                    <Input id="camp-fromname" value={fromName} onChange={(e) => setFromName(e.target.value)} placeholder="Growth Team" />
   429→                  </div>
   430→                  <div className="space-y-1.5">
   431→                    <Label htmlFor="camp-fromemail">From email</Label>
   432→                    <Input id="camp-fromemail" type="email" value={fromEmail} onChange={(e) => setFromEmail(e.target.value)} placeholder="you@company.com" />
   433→                  </div>
   434→                </div>
   435→              </>
   436→            )}
   437→
   438→            {channel === "whatsapp" && (
   439→              <div className="space-y-1.5">
   440→                <Label htmlFor="camp-wa">WhatsApp message</Label>
   441→                <Textarea
   442→                  id="camp-wa"
   443→                  value={bodyText}
   444→                  onChange={(e) => setBodyText(e.target.value)}
   445→                  rows={6}
   446→                  maxLength={1024}
   447→                  placeholder="Hi {{first_name}}! ⚡ Flash sale starts today…"
   448→                />
   449→                <p className="text-xs text-muted-foreground">{bodyText.length}/1024 characters</p>
   450→              </div>
   451→            )}
   452→          </div>
   453→
   454→          <div className="space-y-2">
   455→            {channel === "email" && (
   456→              <>
   457→                <div className="flex items-center justify-between">
   458→                  <Label>Email body (HTML)</Label>
   459→                  <div className="flex gap-1.5">
   460→                    <Button type="button" size="sm" variant={previewMode ? "outline" : "secondary"} onClick={() => setPreviewMode(false)}>
   461→                      HTML
   462→                    </Button>
   463→                    <Button type="button" size="sm" variant={previewMode ? "secondary" : "outline"} onClick={() => setPreviewMode(true)}>
   464→                      <Eye className="mr-1 h-3 w-3" aria-hidden /> Preview
   465→                    </Button>
   466→                  </div>
   467→                </div>
   468→                {previewMode ? (
   469→                  <iframe
   470→                    title="Email preview"
   471→                    srcDoc={previewHtml}
   472→                    className="h-72 w-full rounded-lg border bg-white"
   473→                    sandbox="allow-same-origin"
   474→                  />
   475→                ) : (
   476→                  <Textarea
   477→                    value={bodyHtml}
   478→                    onChange={(e) => setBodyHtml(e.target.value)}
   479→                    rows={12}
   480→                    className="font-mono text-xs"
   481→                    aria-label="Email HTML body"
   482→                  />
   483→                )}
   484→                <p className="flex items-center gap-1 text-xs text-muted-foreground">
   485→                  <Sparkles className="h-3 w-3" aria-hidden /> Open & click tracking is added automatically to every send.
   486→                </p>
   487→              </>
   488→            )}
   489→            {channel === "whatsapp" && (
   490→              <div className="flex h-72 items-center justify-center rounded-lg border bg-stone-50 p-4 dark:bg-stone-900">
   491→                <div className="max-w-[260px] rounded-2xl rounded-br-sm bg-lime-100 p-3 text-sm shadow dark:bg-lime-900">
   492→                  {bodyText.replace(/\{\{\s*first_name\s*\}\}/g, "Alex").replace(/\{\{\s*(\w+)\s*\}\}/g, "—") || <span className="text-stone-400">Message preview…</span>}
   493→                  <p className="mt-1 text-right text-[10px] text-stone-500">delivered ✓✓</p>
   494→                </div>
   495→              </div>
   496→            )}
   497→          </div>
   498→        </div>
   499→
   500→        <DialogFooter>
   501→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   502→          <Button onClick={save} disabled={busy} className="bg-teal-600 hover:bg-teal-700">
   503→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   504→            {campaign ? "Save changes" : "Create campaign"}
   505→          </Button>
   506→        </DialogFooter>
   507→      </DialogContent>
   508→    </Dialog>
   509→  );
   510→}

   512→// ─── Send dialog ─────────────────────────────────────────────
   513→
   514→function SendDialog({
   515→  campaign, onClose, onQueued,
   516→}: {
   517→  campaign: Campaign; onClose: () => void; onQueued: () => void;
   518→}) {
   519→  const { toast } = useToast();
   520→  const [mode, setMode] = useState<"now" | "schedule">("now");
   521→  const [when, setWhen] = useState("");
   522→  const [testTo, setTestTo] = useState("");
   523→  const [busy, setBusy] = useState(false);
   524→  const [testBusy, setTestBusy] = useState(false);
   525→
   526→  async function send() {
   527→    setBusy(true);
   528→    try {
   529→      await api.post(`/api/campaigns/${campaign.id}/send`, {
   530→        mode,
   531→        scheduledAt: mode === "schedule" && when ? new Date(when).toISOString() : undefined,
   532→      });
   533→      onQueued();
   534→    } catch {
   535→      toast({ title: "Send failed", description: "The campaign could not be queued.", variant: "destructive" });
   536→    } finally {
   537→      setBusy(false);
   538→    }
   539→  }
   540→
   541→  async function sendTest() {
   542→    if (!testTo.trim()) return;
   543→    setTestBusy(true);
   544→    try {
   545→      await api.post(`/api/campaigns/${campaign.id}/test`, {
   546→        channel: campaign.channel,
   547→        to: campaign.channel === "email" ? testTo : testTo,
   548→      });
   549→      toast({ title: "Test queued", description: `Sending a test to ${testTo}` });
   550→    } catch {
   551→      toast({ title: "Test send failed", variant: "destructive" });
   552→    } finally {
   553→      setTestBusy(false);
   554→    }
   555→  }
   556→
   557→  return (
   558→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   559→      <DialogContent className="max-w-md">
   560→        <DialogHeader>
   561→          <DialogTitle>Send &quot;{campaign.name}&quot;</DialogTitle>
   562→          <DialogDescription>
   563→            The automation engine dispatches with per-message evidence and retry protection.
   564→          </DialogDescription>
   565→        </DialogHeader>
   566→
   567→        <div className="space-y-4">
   568→          <div className="grid grid-cols-2 gap-2">
   569→            <Button
   570→              variant={mode === "now" ? "default" : "outline"}
   571→              onClick={() => setMode("now")}
   572→              className={mode === "now" ? "bg-teal-600 hover:bg-teal-700" : ""}
   573→            >
   574→              <Send className="mr-1.5 h-4 w-4" aria-hidden /> Send now
   575→            </Button>
   576→            <Button variant={mode === "schedule" ? "default" : "outline"} onClick={() => setMode("schedule")} className={mode === "schedule" ? "bg-teal-600 hover:bg-teal-700" : ""}>
   577→              <Clock className="mr-1.5 h-4 w-4" aria-hidden /> Schedule
   578→            </Button>
   579→          </div>
   580→
   581→          {mode === "schedule" && (
   582→            <div className="space-y-1.5">
   583→              <Label htmlFor="send-when">Date & time</Label>
   584→              <Input id="send-when" type="datetime-local" value={when} onChange={(e) => setWhen(e.target.value)} />
   585→            </div>
   586→          )}
   587→
   588→          <div className="rounded-lg border border-dashed p-3">
   589→            <p className="mb-2 flex items-center gap-1.5 text-xs font-medium text-muted-foreground">
   590→              <FlaskConical className="h-3.5 w-3.5" aria-hidden /> Send a test first
   591→            </p>
   592→            <div className="flex gap-2">
   593→              <Input
   594→                value={testTo}
   595→                onChange={(e) => setTestTo(e.target.value)}
   596→                placeholder={campaign.channel === "email" ? "you@company.com" : "+92300…"}
   597→                aria-label="Test recipient"
   598→              />
   599→              <Button variant="secondary" size="sm" onClick={sendTest} disabled={testBusy || !testTo.trim()}>
   600→                {testBusy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : "Send"}
   601→              </Button>
   602→            </div>
   603→          </div>
   604→        </div>
   605→
   606→        <DialogFooter>
   607→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   608→          <Button onClick={send} disabled={busy || (mode === "schedule" && !when)} className="bg-teal-600 hover:bg-teal-700">
   609→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" aria-hidden />}
   610→            {mode === "now" ? "Send to audience" : "Schedule campaign"}
   611→          </Button>
   612→        </DialogFooter>
   613→      </DialogContent>
   614→    </Dialog>
   615→  );
   616→}
